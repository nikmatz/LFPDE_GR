Δηλώστε πρώτα ότι ο δείκτης είναι ακέραιος: declare(n, integer)\$ assume(n>0)\$.

  3. (γ)Ορίστε H(s) := 1/(s\^{}2 + 0.2*s + 9) και υπολογίστε τα cabs(H(k*%i)), carg(H(k*%i)) για $k=1,3,5$ (Παράδειγμα 6.12). Επαληθεύστε και το $\sum1/n^4$ με sum(1/n\^{}4, n, 1, inf), simpsum.

/* =====================================================================
   Κεφάλαιο 6 — Μιγαδικοί συντελεστές, φάσματα και ταυτότητα Parseval
   Συνοδευτικός κώδικας του βιβλίου «Μετασχηματισμοί Laplace και Fourier,
   Μερικές Διαφορικές Εξισώσεις», Ν. Ματζάκος.
   Εκτέλεση: στο wxMaxima, File > Batch File… ή  batch("maxima_ch06.mac")$
   Δηλώνουμε πρώτα ότι ο δείκτης n είναι ακέραιος (τότε exp(-2 %i n %pi) = 1).
   ===================================================================== */
declare(n, integer)$  assume(n > 0)$

/* ---------- Α. Παράδειγμα 6.2: περιοδική εκθετική ---------- */
c: integrate(exp(-t)*exp(-%i*n*t), t, 0, 2*%pi)/(2*%pi);
ratsimp(demoivre(c) - (1 - exp(-2*%pi))/(2*%pi*(1 + %i*n)));     /* 0 */

/* ---------- Β. Λυμένη Άσκηση 6.1: πλάτος και φάση της θεμελιώδους ---------- */
r: integrate(t*exp(-%i*n*%pi*t), t, 0, 1)/2$
c1: rectform(ev(r, n = 1));
[float(2*cabs(c1)), float(-carg(c1))];                             /* A_1 ≈ 0,377, φ_1 ≈ 2,138 */

/* ---------- Γ. Παράδειγμα 6.6: παλμοί ύψους 2 και διάρκειας 1, T = 3 ---------- */
w0: 2*%pi/3$
cp(k) := rectform(integrate(2*exp(-%i*k*w0*t), t, 0, 1)/3)$
for k: 1 thru 8 do (z: cp(k),
   if float(cabs(z)) > 1e-12 then print(k, "  A =", float(2*cabs(z)), "  φ =", float(-carg(z)))
   else print(k, "  A = 0 (λείπει)"))$

/* ---------- Δ. Παράδειγμα 6.4: από το φάσμα στο σήμα ---------- */
c3: -sqrt(3)/2 + %i/2$
[2*cabs(c3), -carg(c3)];                                            /* A_3 = 2, φ_3 = -5π/6 */
[2*realpart(c3), -2*imagpart(c3)];                                  /* a_3 = -√3, b_3 = -1 */
atan2(-1, -sqrt(3));                                                /* η atan2 βρίσκει το τεταρτημόριο */

/* ---------- Ε. Παράδειγμα 6.12: απόκριση συχνότητας ---------- */
H(s) := 1/(s^2 + 0.2*s + 9)$
for k in [1, 3, 5] do print(k, "  |H| =", cabs(H(k*%i)), "  arg H =", carg(H(k*%i)))$

/* ---------- ΣΤ. Ταυτότητα Parseval: Παράδειγμα 6.9 και Λυμένη Άσκηση 6.3 ---------- */
sum(1/n^4, n, 1, inf), simpsum;                                     /* π^4/90 */
P: integrate(t^2, t, -%pi, %pi)/(2*%pi);                            /* μέση ισχύς της |t|: π²/3 */
float(sqrt((P - (%pi/2)^2 - 8/%pi^2)/(8/%pi^2)));                   /* THD τριγωνικού ≈ 0,121 */

Δραστηριότητα Python: Ο FFT στην πράξη

Το πρόγραμμα δειγματοληπτεί $N=64$ φορές ανά περίοδο το τετραγωνικό κύμα $s(t)=\sgn(\sin t)$, με τιμή $0$ στα άλματα, και εκτιμά τα πλάτη $A_k\approx2|X_k|/N$:

import numpy as np
N = 64
t = 2*np.pi*np.arange(N)/N          # t_m = mT/N, T = 2*pi
x = np.where(t < np.pi, 1.0, -1.0)
x[0] = x[N//2] = 0.0                # mean value at the jumps
X = np.fft.fft(x)
A = 2*np.abs(X[1:10])/N             # A_k = 2|X_k|/N
print(np.round(A, 4))

Η έξοδος $1{,}2722$, $0$, $0{,}4213$, $0$, $0{,}2495,\dots$ προσεγγίζει τις ακριβείς τιμές $\frac{4}{\pi k}=1{,}2732$, $0{,}4244$, $0{,}2546$.

  1. (α)Εξηγήστε με το Θεώρημα 6.5 γιατί το σχετικό σφάλμα αυξάνει με το $k$. Επαναλάβετε με $N=256$ και $N=1024$.
  2. (β)Με sympy.integrate (δείκτης n με integer=True, nonzero=True) υπολογίστε τον $c_n$ της Λυμένης Άσκησης 6.1, σχεδιάστε με matplotlib.pyplot.stem τα $A_n$, $\varphi_n$ και συγκρίνετε με το Σχήμα 6.8.
  3. (γ)Ελέγξτε την ταυτότητα Parseval για την ίδια συνάρτηση: η μέση ισχύς είναι $\frac16$ (scipy.integrate.quad). Πόσοι όροι του $\sum|c_n|^2$ δίνουν ακρίβεια $10^{-4}$;

# =====================================================================
#  Κεφάλαιο 6 — Φάσματα, Parseval και FFT με SymPy, NumPy και SciPy
#  Συνοδευτικός κώδικας του βιβλίου «Μετασχηματισμοί Laplace και Fourier,
#  Μερικές Διαφορικές Εξισώσεις», Ν. Ματζάκος.
#  Τρέχει όπως είναι σε Python 3 (sympy, numpy, scipy, matplotlib)
#  ή απευθείας στη σελίδα του βιβλίου (Pyodide).
# =====================================================================
import sympy as sp
import numpy as np
from scipy.integrate import quad
import matplotlib.pyplot as plt

t = sp.symbols('t', real=True)
n = sp.symbols('n', integer=True, nonzero=True)

# ---------- Α. Ο FFT στην πράξη (Ενότητα 6.6) ----------
for N in (64, 256, 1024):
    tm = 2*np.pi*np.arange(N)/N                 # t_m = mT/N, T = 2π
    xs = np.where(tm < np.pi, 1.0, -1.0)
    xs[0] = xs[N//2] = 0.0                      # μέση τιμή στα άλματα
    A = 2*np.abs(np.fft.fft(xs)[1:10])/N        # A_k ≈ 2|X_k|/N
    print(f"Α. N = {N:4d}:", np.round(A[[0, 2, 4, 6, 8]], 4))
print("   ακριβή 4/(πk):", np.round(4/(np.pi*np.arange(1, 10, 2)), 4))

# ---------- Β. Λυμένη Άσκηση 6.1: c_n, πλάτη και φάσεις ----------
cn = sp.simplify(sp.integrate(t*sp.exp(-sp.I*n*sp.pi*t), (t, 0, 1))/2)
print("\nΒ. c_n =", cn)
k = np.arange(1, 8)
c = np.array([complex(cn.subs(n, int(m))) for m in k])
An, phin = 2*np.abs(c), -np.angle(c)
print("   A_n =", np.round(An, 3))
print("   φ_n =", np.round(phin, 3))
fig, ax = plt.subplots(1, 2, figsize=(8, 2.8))
ax[0].stem(np.r_[0, k], np.r_[0.25, An]); ax[0].set_title("A_n"); ax[0].set_xlabel("n")
ax[1].stem(k, phin); ax[1].set_title("φ_n"); ax[1].set_xlabel("n")
plt.tight_layout(); plt.show()

# ---------- Γ. Ταυτότητα Parseval για την ίδια συνάρτηση ----------
P = quad(lambda s: s*s, 0, 1)[0]/2              # μέση ισχύς: 1/6
for M in (10, 100, 1000):
    m = np.arange(1, M + 1)
    cm = 1j*(-1.0)**m/(2*m*np.pi) - (1 - (-1.0)**m)/(2*m**2*np.pi**2)
    print(f"Γ. M = {M:4d}:  P - Σ|c_n|² = {P - (0.25**2 + 2*np.sum(np.abs(cm)**2)):.2e}")

# ---------- Δ. Παράδειγμα 6.6: παλμοί ύψους 2, διάρκειας 1, περιόδου 3 ----------
k = np.arange(1, 9)
c = 2/(np.pi*k)*np.sin(k*np.pi/3)*np.exp(-1j*k*np.pi/3)
A = 2*np.abs(c); ph = np.where(A > 1e-12, -np.angle(c), np.nan)
print("\nΔ. A_n =", np.round(A, 4))
print("   φ_n/π =", np.round(ph/np.pi, 4))

# ---------- Ε. Παράδειγμα 6.4: από το φάσμα στο σήμα ----------
c1, c3 = 1 - 1j, -np.sqrt(3)/2 + 0.5j
tt = np.linspace(0, 2*np.pi, 7)
f_exp = -0.5 + 2*np.real(c1*np.exp(1j*tt) + c3*np.exp(3j*tt))
f_amp = -0.5 + 2*np.sqrt(2)*np.cos(tt - np.pi/4) + 2*np.cos(3*tt + 5*np.pi/6)
print("\nΕ. μέγιστη διαφορά των δύο μορφών:", np.max(np.abs(f_exp - f_amp)))
print("   atan2(b3, a3) =", np.arctan2(-1, -np.sqrt(3)), " = -5π/6 =", -5*np.pi/6)
